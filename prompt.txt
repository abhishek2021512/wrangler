import io.cdap.wrangler.api.Token;

import java.util.regex.Matcher;
import java.util.regex.Pattern;

/**
 * A token representing a byte size value (e.g., "10KB", "1.5MB").
 */
public class ByteSize extends Token {
    private static final Pattern PATTERN = Pattern.compile("(\\d+(\\.\\d+)?)\\s*([a-zA-Z]+)");
    private static final String[] UNITS = {"B", "KB", "MB", "GB", "TB"};
    private static final long[] MULTIPLIERS = {1L, 1024L, 1024L * 1024, 1024L * 1024 * 1024, 1024L * 1024 * 1024 * 1024};

    private final long bytes;

    /**
     * Constructs a ByteSize token by parsing the input string.
     *
     * @param value the string to parse (e.g., "10KB")
     * @throws IllegalArgumentException if the format is invalid or the unit is unknown
     */
    public ByteSize(String value) {
        super(TokenType.BYTE_SIZE);
        Matcher matcher = PATTERN.matcher(value.trim());
        if (!matcher.matches()) {
            throw new IllegalArgumentException("Invalid byte size format: " + value);
        }
        double number = Double.parseDouble(matcher.group(1));
        String unit = matcher.group(3).toUpperCase();
        int index = -1;
        for (int i = 0; i < UNITS.length; i++) {
            if (UNITS[i].equals(unit)) {
                index = i;
                break;
            }
        }
        if (index == -1) {
            throw new IllegalArgumentException("Unknown byte unit: " + unit);
        }
        if (number < 0) {
            throw new IllegalArgumentException("Byte size cannot be negative: " + value);
        }
        this.bytes = (long) (number * MULTIPLIERS[index]);
    }

    /**
     * Returns the value in bytes.
     *
     * @return the size in bytes
     */
    public long getBytes() {
        return bytes;
    }
}

package io.cdap.wrangler.api.parser;

import io.cdap.wrangler.api.Token;

import java.math.BigInteger;
import java.util.regex.Matcher;
import java.util.regex.Pattern;

/**
 * A token representing a byte size value (e.g., "10KB", "1.5MB").
 */
public class ByteSize extends Token {
    // Matches integers, decimals, or scientific notation followed by optional whitespace and a unit
    private static final Pattern PATTERN = Pattern.compile(
            "^\\s*([-+]?\\d*\\.?\\d+(?:[eE][-+]?\\d+)?)\\s*([a-zA-Z]+)\\s*$");
    private static final String[] UNITS = {"B", "KB", "MB", "GB", "TB"};
    private static final BigInteger[] MULTIPLIERS = {
            BigInteger.ONE,
            BigInteger.valueOf(1024),
            BigInteger.valueOf(1024).pow(2),
            BigInteger.valueOf(1024).pow(3),
            BigInteger.valueOf(1024).pow(4)
    };
    private static final BigInteger MAX_LONG = BigInteger.valueOf(Long.MAX_VALUE);

    private final BigInteger bytes;

    /**
     * Constructs a ByteSize token by parsing the input string.
     *
     * @param value the string to parse (e.g., "10KB", "1.5MB")
     * @throws IllegalArgumentException if the format is invalid, the unit is unknown, or the value causes overflow
     */
    public ByteSize(String value) {
        super(TokenType.BYTE_SIZE);
        if (value == null || value.trim().isEmpty()) {
            throw new IllegalArgumentException("Byte size cannot be null or empty");
        }

        Matcher matcher = PATTERN.matcher(value);
        if (!matcher.matches()) {
            throw new IllegalArgumentException("Invalid byte size format: '" + value + "'. Expected format: number followed by unit (e.g., '10KB')");
        }

        String numberStr = matcher.group(1);
        String unit = matcher.group(2).toUpperCase();

        // Validate unit
        int index = -1;
        for (int i = 0; i < UNITS.length; i++) {
            if (UNITS[i].equals(unit)) {
                index = i;
                break;
            }
        }
        if (index == -1) {
            throw new IllegalArgumentException("Unknown byte unit in '" + value + "': " + unit);
        }

        // Parse number and compute bytes
        try {
            BigInteger number = new BigInteger(numberStr.replace(".", "")); // Handle decimals manually if needed
            BigInteger result = number.multiply(MULTIPLIERS[index]);

            // Check for overflow
            if (result.compareTo(MAX_LONG) > 0) {
                throw new IllegalArgumentException("Byte size too large in '" + value + "': exceeds Long.MAX_VALUE");
            }
            if (result.compareTo(BigInteger.ZERO) < 0) {
                throw new IllegalArgumentException("Byte size cannot be negative in '" + value + "'");
            }

            this.bytes = result;
        } catch (NumberFormatException e) {
            throw new IllegalArgumentException("Invalid number format in '" + value + "': " + numberStr, e);
        }
    }

    /**
     * Returns the size in bytes.
     *
     * @return the size as a long value
     */
    public long getBytes() {
        return bytes.longValue();
    }
}

package io.cdap.wrangler.directives;

import io.cdap.wrangler.api.Arguments;
import io.cdap.wrangler.api.Directive;
import io.cdap.wrangler.api.ExecutorContext;
import io.cdap.wrangler.api.Row;
import io.cdap.wrangler.api.parser.Identifier;
import io.cdap.wrangler.api.parser.TokenType;
import io.cdap.wrangler.api.parser.UsageDefinition;

import java.math.BigInteger;
import java.util.ArrayList;
import java.util.HashMap;
import java.util.List;
import java.util.Map;

/**
 * A directive that computes aggregate statistics (sum, avg, min, max) on specified columns.
 */
public class AggregateStatsDirective implements Directive {
    private Map<String, String> aggregations;

    @Override
    public UsageDefinition define() {
        UsageDefinition.Builder builder = UsageDefinition.builder("aggregate-stats");
        builder.define("aggregations", TokenType.IDENTIFIER);
        return builder.build();
    }

    @Override
    public void initialize(Arguments args) {
        aggregations = new HashMap<>();
        List<Identifier> identifiers = args.value("aggregations");
        for (Identifier id : identifiers) {
            String[] parts = id.value().split(":");
            if (parts.length != 2) {
                throw new IllegalArgumentException("Invalid aggregation format: " + id.value() + ". Expected column:function");
            }
            String column = parts[0];
            String function = parts[1].toLowerCase();
            if (!function.equals("sum") && !function.equals("avg") && !function.equals("min") && !function.equals("max")) {
                throw new IllegalArgumentException("Unsupported function: " + function + ". Supported: sum, avg, min, max");
            }
            aggregations.put(column, function);
        }
    }

    @Override
    public List<Row> execute(List<Row> rows, ExecutorContext context) {
        Map<String, BigInteger> sums = new HashMap<>();
        Map<String, Long> counts = new HashMap<>();
        Map<String, BigInteger> mins = new HashMap<>();
        Map<String, BigInteger> maxs = new HashMap<>();

        // Process each row
        for (Row row : rows) {
            for (Map.Entry<String, String> entry : aggregations.entrySet()) {
                String column = entry.getKey();
                String function = entry.getValue();
                Object value = row.getValue(column);

                BigInteger num = null;
                if (value instanceof ByteSize) {
                    num = BigInteger.valueOf(((ByteSize) value).getBytes());
                } else if (value instanceof TimeDuration) {
                    num = BigInteger.valueOf(((TimeDuration) value).getMilliseconds());
                } else if (value instanceof Number) {
                    num = BigInteger.valueOf(((Number) value).longValue());
                } else {
                    continue; // Skip non-numeric values
                }

                sums.compute(column, (k, v) -> v == null ? num : v.add(num));
                counts.compute(column, (k, v) -> v == null ? 1L : v + 1);
                mins.compute(column, (k, v) -> v == null ? num : v.min(num));
                maxs.compute(column, (k, v) -> v == null ? num : v.max(num));
            }
        }

        // Create result row
        Row result = new Row();
        for (Map.Entry<String, String> entry : aggregations.entrySet()) {
            String column = entry.getKey();
            String function = entry.getValue();
            String outputColumn = column + "_" + function;

            if (function.equals("sum")) {
                BigInteger sum = sums.getOrDefault(column, BigInteger.ZERO);
                result.add(outputColumn, sum.longValue());
            } else if (function.equals("avg")) {
                BigInteger sum = sums.getOrDefault(column, BigInteger.ZERO);
                long count = counts.getOrDefault(column, 0L);
                result.add(outputColumn, count > 0 ? sum.doubleValue() / count : 0.0);
            } else if (function.equals("min")) {
                BigInteger min = mins.get(column);
                result.add(outputColumn, min != null ? min.longValue() : null);
            } else if (function.equals("max")) {
                BigInteger max = maxs.get(column);
                result.add(outputColumn, max != null ? max.longValue() : null);
            }
        }

        return new ArrayList<>(List.of(result));
    }

    @Override
    public void destroy() {
        // No-op
    }
}

Robustness: Uses BigInteger to avoid overflow, handles empty rows, and skips non-numeric values.
Output: Creates a row with results (e.g., file_size_sum, processing_time_avg).

package io.cdap.wrangler.directives;

import io.cdap.wrangler.api.Arguments;
import io.cdap.wrangler.api.Directive;
import io.cdap.wrangler.api.ExecutorContext;
import io.cdap.wrangler.api.Row;
import io.cdap.wrangler.api.parser.ByteSize;
import io.cdap.wrangler.api.parser.ColumnName;
import io.cdap.wrangler.api.parser.Identifier;
import io.cdap.wrangler.api.parser.TimeDuration;
import io.cdap.wrangler.api.parser.TokenType;

import java.math.BigInteger;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;

/**
 * A directive that computes aggregate statistics (sum) on specified columns, outputting byte sizes in MB and time durations in seconds.
 * Recipe format: "aggregate-stats :input_col1 :input_col2 output_col1 output_col2"
 * Example: "aggregate-stats :data_transfer_size :response_time total_size_mb total_time_sec"
 */
public class AggregateStatsDirective implements Directive {
    private String sizeInputColumn;
    private String timeInputColumn;
    private String sizeOutputColumn;
    private String timeOutputColumn;
    private String function; // "sum" or "avg" (for future extension)

    @Override
    public UsageDefinition define() {
        UsageDefinition.Builder builder = UsageDefinition.builder("aggregate-stats");
        builder.define("sizeInput", TokenType.COLUMN_NAME);
        builder.define("timeInput", TokenType.COLUMN_NAME);
        builder.define("sizeOutput", TokenType.IDENTIFIER);
        builder.define("timeOutput", TokenType.IDENTIFIER);
        return builder.build();
    }

    @Override
    public void initialize(Arguments args) {
        sizeInputColumn = ((ColumnName) args.value("sizeInput")).value().substring(1); // Remove ':'
        timeInputColumn = ((ColumnName) args.value("timeInput")).value().substring(1); // Remove ':'
        sizeOutputColumn = ((Identifier) args.value("sizeOutput")).value();
        timeOutputColumn = ((Identifier) args.value("timeOutput")).value();
        function = "sum"; // Default to sum; extend for "avg" in future
    }

    @Override
    public List<Row> execute(List<Row> rows, ExecutorContext context) {
        BigInteger sizeSumBytes = BigInteger.ZERO;
        BigInteger timeSumMillis = BigInteger.ZERO;
        long count = 0;

        // Process each row
        for (Row row : rows) {
            // Handle size column
            Object sizeValue = row.getValue(sizeInputColumn);
            if (sizeValue != null) {
                BigInteger bytes;
                if (sizeValue instanceof ByteSize) {
                    bytes = BigInteger.valueOf(((ByteSize) sizeValue).getBytes());
                } else if (sizeValue instanceof Number) {
                    bytes = BigInteger.valueOf(((Number) sizeValue).longValue());
                } else {
                    continue; // Skip non-numeric values
                }
                sizeSumBytes = sizeSumBytes.add(bytes);
            }

            // Handle time column
            Object timeValue = row.getValue(timeInputColumn);
            if (timeValue != null) {
                BigInteger millis;
                if (timeValue instanceof TimeDuration) {
                    millis = BigInteger.valueOf(((TimeDuration) timeValue).getMilliseconds());
                } else if (timeValue instanceof Number) {
                    millis = BigInteger.valueOf(((Number) timeValue).longValue());
                } else {
                    continue; // Skip non-numeric values
                }
                timeSumMillis = timeSumMillis.add(millis);
            }

            count++;
        }

        // Create result row
        Row result = new Row();
        // Convert size to MB (1 MB = 1024 * 1024 bytes)
        double sizeInMB = sizeSumBytes.doubleValue() / (1024 * 1024);
        result.add(sizeOutputColumn, sizeInMB);

        // Convert time to seconds (1 second = 1000 milliseconds)
        double timeInSeconds = timeSumMillis.doubleValue() / 1000.0;
        result.add(timeOutputColumn, timeInSeconds);

        return new ArrayList<>(List.of(result));
    }

    @Override
    public void destroy() {
        // No-op
    }
}
